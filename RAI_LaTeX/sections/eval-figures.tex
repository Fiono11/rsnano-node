% The evaluation's figures; data in sections/eval-data/wan (tools/wan_figures.py,
% run_sweep.py --set final-wan-2000 across six regions, seven with the committee change).
\pgfplotsset{
  paperaxis/.style={tick label style={font=\scriptsize},label style={font=\scriptsize},title style={font=\scriptsize,yshift=-1.5mm},
    ymajorgrids,grid style={gray!25},unbounded coords=jump,every axis plot/.append style={line width=0.6pt}},
  timeaxis/.style={paperaxis,xmin=0,ymin=0,ymax=4000,ytick={0,2000,4000},yticklabels={0,2k,4k},
    restrict y to domain*=0:4000,no markers},
  scenarioticks/.style={xmin=-0.5,xmax=5.5,xtick={0,...,5},
    xticklabels={no forks,5\%,10\%,{no forks,\\ Byz.},{5\%,\\ Byz.},{10\%,\\ Byz.}},x tick label style={align=center}},
  latencyaxis/.style={paperaxis,log origin=infty,ymin=80,ymax=100000,
    ytick={100,1000,10000,100000},yticklabels={100,1k,10k,100k},ylabel={latency (ms)}},
  goodputaxis/.style={paperaxis,ymin=0,ymax=2100,ytick={0,500,1000,1500,2000},yticklabels={0,500,1k,1.5k,2k},
    ylabel={confirmed/s}},
}
\colorlet{cRaiLight}{cRai!45}
\newcommand{\matrixdata}[1]{sections/eval-data/#1.dat}
% Latency bars of one system: #1 data file, #2 color, #3 shift, #4 bar
% width; p95 outlined, p50 filled, each with the range over the reps
\newcommand{\latbars}[4]{%
  \addplot[ybar,bar width=#4,bar shift=#3,draw=#2,fill=white,line width=0.5pt,forget plot,
    error bars/.cd,y dir=both,y explicit,error bar style={#2,line width=0.4pt}]
    table[x=x,y=p95,y error plus expr=\thisrow{p95hi}-\thisrow{p95},y error minus expr=\thisrow{p95}-\thisrow{p95lo}] {\matrixdata{#1}};
  \addplot[ybar,bar width=#4,bar shift=#3,draw=#2,fill=#2,line width=0.5pt,area legend,
    error bars/.cd,y dir=both,y explicit,error bar style={black!60,line width=0.4pt}]
    table[x=x,y=p50,y error plus expr=\thisrow{p50hi}-\thisrow{p50},y error minus expr=\thisrow{p50}-\thisrow{p50lo}] {\matrixdata{#1}};
}
% Goodput bars of one system, with the range over the reps
\newcommand{\gputbars}[4]{%
  \addplot[ybar,bar width=#4,bar shift=#3,draw=#2,fill=#2!70,line width=0.5pt,
    error bars/.cd,y dir=both,y explicit,error bar style={black!70,line width=0.4pt}]
    table[x=x,y=gput,y error plus expr=\thisrow{gputhi}-\thisrow{gput},y error minus expr=\thisrow{gput}-\thisrow{gputlo}] {\matrixdata{#1}};
}
% One timeline panel: #1 the axis top, #2 data directory, #3 run name, #4
% color. The close windows are shaded, the non-fork confirmations per second
% drawn on top, the end of the load dashed
\newcommand{\timeline}[4][4000]{%
  \addplot[fill=#4!18,draw=none] table[x=t,y expr=\thisrow{y}*#1] {\matrixdata{#2/closes-#3}} \closedcycle;
  \addplot[color=#4,line width=0.7pt] table[x=t,y=confirmed] {\matrixdata{#2/timeline-#3}};
  \addplot[black!55,densely dashed,line width=0.5pt] table[x=t,y expr=\thisrow{y}*#1] {\matrixdata{#2/loadend-#3}};
}
