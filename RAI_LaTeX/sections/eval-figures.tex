% The evaluation's figures; data in sections/eval-data/paper
% (tools/paper_figures.py from the 'paper' set of tools/rai/run_sweep.py,
% offered load 2,000 blocks/s).
\pgfplotsset{
  paperaxis/.style={tick label style={font=\scriptsize},label style={font=\scriptsize},title style={font=\scriptsize,yshift=-1.5mm},
    ymajorgrids,grid style={gray!25},unbounded coords=jump,every axis plot/.append style={line width=0.6pt}},
  timeaxis/.style={paperaxis,xmin=0,xmax=36,ymin=0,ymax=4000,ytick={0,2000,4000},yticklabels={0,2k,4k},
    restrict y to domain*=0:4000,no markers},
}
\newcommand{\paperdata}[1]{sections/eval-data/paper/#1.dat}
% One timeline panel: #1 scenario, #2 system, #3 color. The close windows
% are shaded, the system's non-fork confirmations per second drawn on top,
% the end of the load dashed
\newcommand{\timeline}[3]{%
  \addplot[fill=#3!18,draw=none] table[x=t,y expr=\thisrow{y}*4000] {\paperdata{closes-#1-#2}} \closedcycle;
  \addplot[color=#3,line width=0.7pt] table[x=t,y=confirmed] {\paperdata{timeline-#1-#2}};
  \addplot[black!55,densely dashed,line width=0.5pt] table[x=t,y expr=\thisrow{y}*4000] {\paperdata{loadend-#1-#2}};
}

\centering
\begin{tikzpicture}
\begin{groupplot}[group style={group size=2 by 3,horizontal sep=5mm,vertical sep=3mm,xticklabels at=edge bottom,yticklabels at=edge left},
  timeaxis,width=0.52\textwidth,height=2.7cm]
\nextgroupplot[title={10\% forks},ylabel={RAI\\ confirmed/s},ylabel style={align=center}]

\timeline{fork5-byzlead-grace}{develop}{cNano}
\end{groupplot}
\end{tikzpicture}
\caption{Non-fork blocks confirmed per second at an offered load of 2{,}000 blocks/s, with 8\,s epochs; shaded: an epoch is closing (from the first validator leaving it to the last one seeing its checkpoint decided); dashed: the end of the load, after which there is little left to confirm. Pause confirms nothing until the checkpoint is decided, then catches up in a burst (bursts above 4{,}000/s are cut off); RAI keeps confirming through every close. A Byzantine validator that leads the first round of every close and stays silent makes each close wait for the round timeout, which stops pause for longer and does not stop RAI. Nano has no epochs and never stops, but with 10\% forks it confirms below the offered load and finishes its backlog in a trickle after the load ends. One run each.}
\label{fig:eval-timeline}
\end{figure*}

% Bars of two values per system: #1 data file prefix, #2 system, #3 color,
% #4 shift, #5 the larger value (outline), #6 the smaller one (filled)
\newcommand{\pairbars}[6]{%
  \addplot[ybar,bar width=7pt,bar shift=#4,draw=#3,fill=white,line width=0.6pt,forget plot] table[x=x,y=#5] {\paperdata{#1-#2}};
  \addplot[ybar,bar width=7pt,bar shift=#4,draw=#3,fill=#3,line width=0.6pt,area legend] table[x=x,y=#6] {\paperdata{#1-#2}};
}

  xticklabels={no forks,5\% forks,10\% forks,{5\% forks,\\ Byz.\ leader}},x tick label style={align=center}}}
