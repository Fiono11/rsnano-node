\newcommand{\latplot}[3]{% #1 arm style, #2 x offset, #3 column (p50 or p95)
  \addplot[#1] table[x expr=\thisrow{x}+#2,y=#3,y error plus expr=\thisrow{#3_hi}-\thisrow{#3},y error minus expr=\thisrow{#3}-\thisrow{#3_lo}] {sections/eval-data/summary-#1.dat};}
