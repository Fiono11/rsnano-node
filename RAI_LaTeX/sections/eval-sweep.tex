\pgfplotsset{
  sweepaxis/.style={tick label style={font=\scriptsize},label style={font=\scriptsize},title style={font=\scriptsize,yshift=-1.5mm},
    ymode=log,ymin=60,ymax=20000,ytick={100,200,500,1000,2000,5000,10000},yticklabels={100,200,500,1k,2k,5k,10k},
    xmin=0,xmax=3000,xtick={0,1000,2000,3000},xticklabels={0,1k,2k,3k},ymajorgrids,xmajorgrids,grid style={gray!25},
    unbounded coords=jump,every axis plot/.append style={line width=0.6pt}},
  s50/.style={mark=*,mark size=1.1pt},
  s95/.style={densely dashed,mark=o,mark size=1.1pt},
}
% One system's curves in a throughput-latency panel: #1 data file, #2 color
\newcommand{\sweepcurves}[2]{%
  \addplot[color=#2,s50] table[x=throughput,y=p50] {#1};
  \addplot[color=#2,s95] table[x=throughput,y=p95] {#1};
}

  \draw[black,densely dashed,line width=0.6pt] (7.6,0) -- (8.1,0) node[right] {p95};
  \end{tikzpicture}}
